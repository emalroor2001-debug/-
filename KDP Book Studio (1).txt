// 1. PROJECTS & PAGES TAB
// ==========================================
function ProjectsTab({ projects, setProjects, selectedProjectId, setSelectedProjectId, activeProject, lang, darkMode, copyToClipboard }) {
  const [editingPage, setEditingPage] = useState(null);
  const [newPageTitle, setNewPageTitle] = useState('');

  const handleAddPage = () => {
    if (!activeProject) return;
    const nextNum = activeProject.pages.length + 1;
    const newPage = {
      id: `page-${Date.now()}`,
      pageNum: nextNum,
      title: newPageTitle || `صفحة ${nextNum} - Page ${nextNum}`,
      prompt: `Coloring book page for children, clean vector line art, no shading, subject: ${newPageTitle || 'cute animal adventure'}, white background`,
      status: 'Draft'
    };
    
    const updatedProjects = projects.map(p => {
      if (p.id === activeProject.id) {
        return { ...p, pages: [...p.pages, newPage], updatedAt: new Date().toISOString().split('T')[0] };
      }
      return p;
    });
    setProjects(updatedProjects);
    setNewPageTitle('');
  };

  const handleDeletePage = (pageId) => {
    const updatedProjects = projects.map(p => {
      if (p.id === activeProject.id) {
        const filtered = p.pages.filter(pg => pg.id !== pageId);
        // re-index page numbers
        const reindexed = filtered.map((pg, idx) => ({ ...pg, pageNum: idx + 1 }));
        return { ...p, pages: reindexed };
      }
      return p;
    });
    setProjects(updatedProjects);
  };

  const handleStatusChange = (newStatus) => {
    setProjects(projects.map(p => p.id === activeProject.id ? { ...p, status: newStatus } : p));
  };

  return (
    <div className="grid grid-cols-1 lg:grid-cols-12 gap-6">
      {/* SIDEBAR: Project List */}
      <div className="lg:col-span-4 space-y-4">
        <div className={`p-4 rounded-2xl border ${darkMode ? 'bg-slate-900/60 border-slate-800' : 'bg-white border-slate-200'}`}>
          <div className="flex items-center justify-between mb-3">
            <h2 className="font-bold text-base flex items-center gap-2">
              <Book className="w-5 h-5 text-indigo-400" />
              <span>{lang === 'ar' ? 'مشاريعك' : 'Your Projects'}</span>
            </h2>
            <span className="text-xs px-2 py-0.5 rounded-full bg-slate-800 text-slate-400 font-mono">{projects.length}</span>
          </div>

          <div className="space-y-2.5">
            {projects.map((proj) => {
              const isSelected = proj.id === selectedProjectId;
              return (
                <div
                  key={proj.id}
                  onClick={() => setSelectedProjectId(proj.id)}
                  className={`p-3.5 rounded-xl cursor-pointer transition-all border ${
                    isSelected
                      ? 'bg-indigo-600/10 border-indigo-500/50 shadow-md'
                      : darkMode ? 'bg-slate-800/30 border-slate-800 hover:border-slate-700' : 'bg-slate-100/70 border-slate-200 hover:border-slate-300'
                  }`}
                >
                  <div className="flex items-start justify-between">
                    <h3 className={`font-semibold text-sm line-clamp-1 ${isSelected ? 'text-indigo-400' : ''}`}>
                      {proj.title}
                    </h3>
                    <span className={`text-[10px] font-medium px-2 py-0.5 rounded-full ${
                      proj.status === 'Ready for KDP' ? 'bg-emerald-500/10 text-emerald-400 border border-emerald-500/20' :
                      proj.status === 'In Progress' ? 'bg-amber-500/10 text-amber-400 border border-amber-500/20' : 'bg-slate-500/10 text-slate-400'
                    }`}>
                      {proj.status}
                    </span>
                  </div>

                  <div className="mt-2.5 flex items-center justify-between text-xs text-slate-400 font-mono">
                    <span>{proj.type} • {proj.trimSize}</span>
                    <span>{proj.pages.length} {lang === 'ar' ? 'صفحة' : 'Pages'}</span>
                  </div>
                </div>
              );
            })}
          </div>
        </div>

        {/* Quick Project Overview Card */}
        {activeProject && (
          <div className={`p-4 rounded-2xl border ${darkMode ? 'bg-slate-900/60 border-slate-800' : 'bg-white border-slate-200'} space-y-3`}>
            <h3 className="text-xs font-semibold uppercase tracking-wider text-slate-400">
              {lang === 'ar' ? 'تفاصيل الكتاب المختارة' : 'Selected Book Details'}
            </h3>
            
            <div className="grid grid-cols-2 gap-2 text-xs">
              <div className={`p-2.5 rounded-lg ${darkMode ? 'bg-slate-800/50' : 'bg-slate-100'}`}>
                <span className="text-slate-400 block">{lang === 'ar' ? 'القياس (Trim)' : 'Trim Size'}</span>
                <span className="font-semibold text-sm">{activeProject.trimSize} in</span>
              </div>
              <div className={`p-2.5 rounded-lg ${darkMode ? 'bg-slate-800/50' : 'bg-slate-100'}`}>
                <span className="text-slate-400 block">{lang === 'ar' ? 'الفئة العمرية' : 'Age Group'}</span>
                <span className="font-semibold text-sm">{activeProject.targetAge}</span>
              </div>
              <div className={`p-2.5 rounded-lg ${darkMode ? 'bg-slate-800/50' : 'bg-slate-100'}`}>
                <span className="text-slate-400 block">{lang === 'ar' ? 'عدد الصفحات النهائي' : 'Target Pages'}</span>
                <span className="font-semibold text-sm">{activeProject.pageCount} pgs</span>
              </div>
              <div className={`p-2.5 rounded-lg ${darkMode ? 'bg-slate-800/50' : 'bg-slate-100'}`}>
                <span className="text-slate-400 block">{lang === 'ar' ? 'نوع الورق' : 'Paper Type'}</span>
                <span className="font-semibold text-sm capitalize">{activeProject.paperType}</span>

              <label className="text-xs text-slate-400 mb-1 block">{lang === 'ar' ? 'حالة المشروع الحالية:' : 'Current Status:'}</label>
              <select
                value={activeProject.status}
                onChange={(e) => handleStatusChange(e.target.value)}
                className={`w-full p-2 text-xs rounded-lg font-medium border ${

                <option value="Draft">Draft (مسودة)</option>
                <option value="In Progress">In Progress (قيد العمل)</option>
                <option value="Ready for KDP">Ready for KDP (جاهز للنشر)</option>

            <div>

              <p className="text-xs text-slate-400">{lang === 'ar' ? 'هيكلة الصفحات ورسومات التلوين (Single-Sided Page Planner)' : 'Page Outline & Illustration Blueprint'}</p>

// 3. BOOK SPECIFICATION & COVER CALCULATOR
// ==========================================
function CalculatorTab({ activeProject, lang, darkMode, copyToClipboard }) {
  const [trimSize, setTrimSize] = useState('8.5x11');
  const [pageCount, setPageCount] = useState(activeProject?.pageCount || 40);
  const [paperColor, setPaperColor] = useState('white');

  // KDP Standard Calculations
  const calculatedSpecs = useMemo(() => {
    let width = 8.5;
    let height = 11;

    if (trimSize === '8x10') { width = 8; height = 10; }
    if (trimSize === '6x9') { width = 6; height = 9; }
    if (trimSize === '8.25x8.25') { width = 8.25; height = 8.25; }

    // Spine width calculation based on Amazon KDP paper thickness formula
    // White paper: pageCount * 0.002252 in
    // Cream paper: pageCount * 0.0025 in
    const thicknessFactor = paperColor === 'white' ? 0.002252 : 0.0025;
    const spineWidth = pageCount * thicknessFactor;

    // Bleed calculation (0.125 inches on top, bottom, and outside margin)
    const bleedMargin = 0.125;
    const singlePageWidthWithBleed = width + bleedMargin;
    const singlePageHeightWithBleed = height + (bleedMargin * 2);

    // Total Full Cover Dimensions (Front + Back + Spine + Bleed)
    const totalCoverWidth = (width * 2) + spineWidth + (bleedMargin * 2);
    const totalCoverHeight = height + (bleedMargin * 2);

    return {
      width,
      height,
      spineWidth: spineWidth.toFixed(4),
      totalCoverWidth: totalCoverWidth.toFixed(3),
      totalCoverHeight: totalCoverHeight.toFixed(3),
      singlePageWidthWithBleed: singlePageWidthWithBleed.toFixed(3),
      singlePageHeightWithBleed: singlePageHeightWithBleed.toFixed(3)
    };
  }, [trimSize, pageCount, paperColor]);

  return (
    <div className="grid grid-cols-1 lg:grid-cols-12 gap-6">
      {/* INPUT FORM */}
      <div className={`lg:col-span-5 p-5 rounded-2xl border ${darkMode ? 'bg-slate-900/60 border-slate-800' : 'bg-white border-slate-200'} space-y-4`}>
        <div className="flex items-center gap-2 border-b border-slate-800 pb-3">
          <Calculator className="w-5 h-5 text-indigo-400" />
          <h2 className="font-bold text-base">{lang === 'ar' ? 'حاسبة أبعاد غلاف وصفحات KDP' : 'KDP Cover & Trim Calculator'}</h2>
        </div>

        <div>
          <label className="text-xs font-semibold text-slate-400 mb-1 block">{lang === 'ar' ? 'مقاس الكتاب (Trim Size):' : 'Trim Size:'}</label>
          <select
            value={trimSize}
            onChange={(e) => setTrimSize(e.target.value)}
            className={`w-full p-2.5 text-xs sm:text-sm rounded-xl border ${darkMode ? 'bg-slate-800 border-slate-700 text-slate-200' : 'bg-slate-100 border-slate-300'}`}
          >
            <option value="8.5x11">8.5 x 11 in (الأكثر مبيعاً للتلوين - Standard)</option>
            <option value="8x10">8 x 10 in (مناسب لقصص الأطفال)</option>
            <option value="8.25x8.25">8.25 x 8.25 in (مربع - Square)</option>
            <option value="6x9">6 x 9 in (روايات وقصص جيب)</option>
          </select>
        </div>

        <div>
          <label className="text-xs font-semibold text-slate-400 mb-1 block">{lang === 'ar' ? 'إجمالي عدد الصفحات:' : 'Page Count:'}</label>
          <input
            type="number"
            min={24}
            max={800}
            value={pageCount}
            onChange={(e) => setPageCount(parseInt(e.target.value) || 24)}
            className={`w-full p-2.5 text-xs sm:text-sm rounded-xl border ${darkMode ? 'bg-slate-800 border-slate-700 text-slate-200' : 'bg-slate-100 border-slate-300'}`}
          />
          <p className="text-[11px] text-slate-400 mt-1">{lang === 'ar' ? 'أدنى عدد صفحات لـ KDP هو 24 صفحة' : 'Minimum 24 pages required by KDP'}</p>
        </div>

          <label className="text-xs font-semibold text-slate-400 mb-1 block">{lang === 'ar' ? 'نوع الورق:' : 'Paper Type:'}</label>
          <div className="grid grid-cols-2 gap-2">
            <button
              onClick={() => setPaperColor('white')}
              className={`p-2.5 rounded-xl text-xs font-medium border transition ${
                paperColor === 'white' ? 'bg-indigo-600/20 border-indigo-500 text-indigo-300' : darkMode ? 'bg-slate-800 border-slate-700 text-slate-400' : 'bg-slate-100'
              }`}
            >
              {lang === 'ar' ? 'ورق أبيض (White)' : 'White Paper'}
            </button>
            <button
              onClick={() => setPaperColor('cream')}
              className={`p-2.5 rounded-xl text-xs font-medium border transition ${
                paperColor === 'cream' ? 'bg-indigo-600/20 border-indigo-500 text-indigo-300' : darkMode ? 'bg-slate-800 border-slate-700 text-slate-400' : 'bg-slate-100'
              }`}
            >
              {lang === 'ar' ? 'ورق كريمي (Cream)' : 'Cream Paper'}
            </button>
          </div>
        </div>
      </div>

      {/* CALCULATED RESULTS & VISUALIZER */}
      <div className="lg:col-span-7 space-y-4">
        <div className={`p-5 rounded-2xl border ${darkMode ? 'bg-slate-900/60 border-slate-800' : 'bg-white border-slate-200'} space-y-4`}>

              <span className="text-slate-400 text-xs block">{lang === 'ar' ? 'سُمْك الكعب (Spine)' : 'Spine Width'}</span>
              <span className="font-mono text-base font-bold text-amber-400">{calculatedSpecs.spineWidth} in</span>
            </div>
          </div>

          {/* Graphical Blueprint Preview */}
          <div className="mt-4 p-4 rounded-xl bg-slate-950 border border-slate-800 flex flex-col items-center justify-center">
            <span className="text-xs text-slate-400 mb-2 font-mono">{lang === 'ar' ? 'رسم توضيحي لأبعاد الغلاف:' : 'Cover Layout Dimensions Blueprint:'}</span>
            <div className="w-full max-w-md aspect-[16/10] border-2 border-dashed border-indigo-500/40 rounded-lg p-2 flex gap-1 relative bg-slate-900/80">

                Back Cover
              </div>

          </div>

          <div className="p-4 rounded-xl bg-emerald-500/10 border border-emerald-500/20 text-xs space-y-1">

  const [trimSize, setTrimSize] = useState('8.5x11');

  const [pageCount, setPageCount] = useState(40);

    e.preventDefault();
    if (!title.trim()) return;

      id: `proj-${Date.now()}`,
      title,
      type,
      trimSize,
      targetAge,
      pageCount: Number(pageCount),
      paperType: 'white',
      status: 'Draft',
      updatedAt: new Date().toISOString().split('T')[0],

        { id: `p-${Date.now()}-1`, pageNum: 1, title: 'Title Page', prompt: `Coloring book cover outline for ${title}, clean line art`, status: 'Draft' }

      titleKdp: title,
      subtitleKdp: '',
      descriptionHtml: '',
      categories: []

              placeholder="e.g. Magical Unicorns Coloring Book"
